board (3, 3)
players 2
piece X
piece O

// ── Player piece assignment ───────────────────────────────────
player p1 has 5 piece X
player p2 has 4 piece O

// ── Common rules for all pieces ──────────────────────────────
gamerules pos piece {
    !occupied(pos)
}

// ── Win condition ─────────────────────────────────────────────
win when positions p {
    occupied((p, p)) and piece((p, p)) == piece(offset((p, p), vertical, 1))
    or
    occupied((p, p)) and piece((p, p)) == piece(offset((p, p), horizontal, 1))
    or
    occupied((p, p)) and piece((p, p)) == piece(offset((p, p), diagonalAsc, 1))
    or
    occupied((p, p)) and piece((p, p)) == piece(offset((p, p), diagonalDesc, 1))
}

// ── Draw condition ────────────────────────────────────────────
draw when global {
    count(X) + count(O) == 9
}

// ── Tests ─────────────────────────────────────────────────────
test {
    place piece X at (1, 1);
    assert(occupied((1, 1)));

    place piece O at (2, 2);
    assert(occupied((2, 2)));

    place piece X at (1, 2);
    place piece O at (3, 3);
    place piece X at (1, 3);
    assert win
}
